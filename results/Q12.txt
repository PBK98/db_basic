-- Q12 | 서브쿼리: 전체 평균보다 조리 시간이 긴 요리를 찾습니다.
SELECT name, cooking_minutes FROM recipe
WHERE cooking_minutes > (SELECT AVG(cooking_minutes) FROM recipe)
ORDER BY cooking_minutes DESC, recipe_id;
