-- Q05 | INNER JOIN: 요리 이름 옆에 분류 이름을 표시합니다.
SELECT r.name AS recipe_name, c.name AS cuisine_name
FROM recipe AS r INNER JOIN cuisine AS c ON r.cuisine_id = c.cuisine_id
ORDER BY r.recipe_id;
