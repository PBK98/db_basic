-- Q03 | 기본 조회: 쉬운 요리 중 조리 시간이 짧은 5개를 찾습니다.
-- LIMIT는 SQLite에서 지원하는 행 수 제한 문법입니다.
SELECT name, difficulty, cooking_minutes FROM recipe
WHERE difficulty = '쉬움' ORDER BY cooking_minutes, recipe_id LIMIT 5;
