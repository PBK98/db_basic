-- Q09 | 집계: 분류별 요리 개수를 셉니다.
SELECT c.name, COUNT(r.recipe_id) AS recipe_count
FROM cuisine AS c LEFT JOIN recipe AS r ON c.cuisine_id = r.cuisine_id
GROUP BY c.cuisine_id, c.name ORDER BY c.cuisine_id;
