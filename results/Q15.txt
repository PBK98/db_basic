-- Q15 | 인덱스: 특정 회원의 리뷰를 찾거나 조인할 때 검색을 돕습니다.
-- 회원별 리뷰 조회에 사용하는 member_id에 인덱스를 만듭니다.
CREATE INDEX IF NOT EXISTS idx_review_member_id ON review(member_id);
